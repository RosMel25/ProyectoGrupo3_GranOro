CREATE TABLE AUDITORIA_VEHICULOS (
    ID_AUDITORIA         NUMBER GENERATED BY DEFAULT AS IDENTITY,
    PLACA_VEHICULO       VARCHAR2(20),
    ACCION               VARCHAR2(10),
    FECHA_ACCION         DATE,
    USUARIO_BD           VARCHAR2(30)
);
/
CREATE OR REPLACE TRIGGER TRG_AUDIT_VEHICULOS
AFTER INSERT OR UPDATE OR DELETE ON Vehiculos
FOR EACH ROW
BEGIN
    IF INSERTING THEN
        INSERT INTO AUDITORIA_VEHICULOS (PLACA_VEHICULO, ACCION, FECHA_ACCION, USUARIO_BD)
        VALUES (:NEW.Placa_Vehiculo, 'INSERT', SYSDATE, USER);

    ELSIF UPDATING THEN
        INSERT INTO AUDITORIA_VEHICULOS (PLACA_VEHICULO, ACCION, FECHA_ACCION, USUARIO_BD)
        VALUES (:NEW.Placa_Vehiculo, 'UPDATE', SYSDATE, USER);

    ELSIF DELETING THEN
        INSERT INTO AUDITORIA_VEHICULOS (PLACA_VEHICULO, ACCION, FECHA_ACCION, USUARIO_BD)
        VALUES (:OLD.Placa_Vehiculo, 'DELETE', SYSDATE, USER);
    END IF;
END;
