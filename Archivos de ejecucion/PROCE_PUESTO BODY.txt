create or replace NONEDITIONABLE PACKAGE BODY PROCE_PUESTO AS

  PROCEDURE AGREGAR_PUESTO (pID IN NUMBER, pNOMBRE_PUESTO IN VARCHAR2,pMIN_SALARIO IN NUMBER,pMAX_SALARIO IN NUMBER) AS
  BEGIN    
    INSERT INTO estudiante.PUESTO (ID_PUESTO,NOMBRE_PUESTO,MIN_SALARIO,MAX_SALARIO)
    VALUES (pID,pnombre_puesto,pmin_salario,pmax_salario);
    COMMIT;
  END AGREGAR_PUESTO;

  PROCEDURE EDITAR_PUESTO (pID IN NUMBER, pNOMBRE_PUESTO IN VARCHAR2,pMIN_SALARIO IN NUMBER,pMAX_SALARIO IN NUMBER) AS
  BEGIN
    UPDATE estudiante.PUESTO
    SET ID_PUESTO = pID, NOMBRE_PUESTO = pnombre_puesto, MIN_SALARIO = pmin_salario, MAX_SALARIO = pmax_salario
    WHERE ID_PUESTO = pID;
    COMMIT;
  END EDITAR_PUESTO;

  PROCEDURE ELIMINAR_PUESTO (pID IN NUMBER) AS
  BEGIN
    DELETE FROM estudiante.PUESTO
  WHERE ID_PUESTO = pID;
  COMMIT;
  END ELIMINAR_PUESTO;

  PROCEDURE BUSCAR_PUESTO (cur_PUESTOS OUT SYS_REFCURSOR) AS
  BEGIN
    OPEN cur_puestos FOR SELECT * FROM estudiante.PUESTO;
  END BUSCAR_PUESTO;
  
  --CURSOR
  PROCEDURE MOSTRAR_PUESTOS_CANTIDAD(cur_puestos OUT SYS_REFCURSOR) AS
BEGIN
    OPEN cur_puestos FOR
        SELECT NOMBRE_PUESTO, CANTIDAD_EMPLEADOS
        FROM VISTA_PUESTOS_EMPLEADOS;
END MOSTRAR_PUESTOS_CANTIDAD;
  
  --MÁS PROCEDIMIENTOS
  -- Aumentar el salario máximo de todos los puestos en un porcentaje dado
  PROCEDURE AUMENTAR_SALARIOS_MAXIMOS(pPORCENTAJE IN NUMBER) AS
BEGIN
    UPDATE PUESTO
    SET MAX_SALARIO = MAX_SALARIO + (MAX_SALARIO * pPORCENTAJE / 100);
    COMMIT;
END AUMENTAR_SALARIOS_MAXIMOS;

-- Establecer el mismo salario mínimo para todos los puestos
PROCEDURE UNIFICAR_SALARIOS_MINIMOS(pSALARIO IN NUMBER) AS
BEGIN
    UPDATE PUESTO
    SET MIN_SALARIO = pSALARIO;
    COMMIT;
END UNIFICAR_SALARIOS_MINIMOS;

-- Mostrar puestos con más de X empleados usando cursor OUT
PROCEDURE MOSTRAR_PUESTOS_CON_MUCHOS_EMPLEADOS(pMIN_CANTIDAD IN NUMBER, cur_PUESTOS OUT SYS_REFCURSOR) AS
BEGIN
    OPEN cur_PUESTOS FOR
        SELECT p.ID_PUESTO, p.NOMBRE_PUESTO, COUNT(e.IDENTIFICACION_EMPLEADO) AS CANTIDAD_EMPLEADOS
        FROM PUESTO p
        JOIN EMPLEADOS e ON p.ID_PUESTO = e.ID_PUESTO
        GROUP BY p.ID_PUESTO, p.NOMBRE_PUESTO
        HAVING COUNT(e.IDENTIFICACION_EMPLEADO) > pMIN_CANTIDAD;
END MOSTRAR_PUESTOS_CON_MUCHOS_EMPLEADOS;

END PROCE_PUESTO;
