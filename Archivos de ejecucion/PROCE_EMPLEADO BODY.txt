create or replace NONEDITIONABLE PACKAGE BODY PROCE_EMPLEADO AS

  PROCEDURE AGREGAR_EMPLEADO (pID IN NUMBER, pNOMBRE IN VARCHAR2,pAPELLIDO IN VARCHAR2, pEMAIL IN VARCHAR2,pTELEFONO IN NUMBER,pSALARIO IN NUMBER,pIDPuesto IN NUMBER,pIDFinca IN NUMBER) AS
  BEGIN
    INSERT INTO estudiante.EMPLEADOS (IDENTIFICACION_EMPLEADO,NOMBRE_EMPLEADO,APELLIDO_EMPLEADO,EMAIL,TELEFONO,SALARIO,ID_PUESTO,NO_FINCA)
    VALUES (pID,pnombre,papellido,pemail,ptelefono,psalario,pidpuesto,pidfinca);
    COMMIT;
  END AGREGAR_EMPLEADO;

  PROCEDURE BUSCAR_EMPLEADO (cur_EMPLEADOS OUT SYS_REFCURSOR) AS
  BEGIN
    OPEN cur_empleados FOR SELECT * FROM estudiante.EMPLEADOS;
  END BUSCAR_EMPLEADO;
  
   PROCEDURE EDITAR_EMPLEADO (pID IN NUMBER, pNOMBRE IN VARCHAR2,pAPELLIDO IN VARCHAR2, pEMAIL IN VARCHAR2,pTELEFONO IN NUMBER,pSALARIO IN NUMBER,pIDPuesto IN NUMBER,pIDFinca IN NUMBER) AS
  BEGIN
    UPDATE estudiante.EMPLEADOS 
    SET IDENTIFICACION_EMPLEADO = pID, NOMBRE_EMPLEADO = pNOMBRE, APELLIDO_EMPLEADO = pAPELLIDO, EMAIL = pEMAIL, TELEFONO = pTELEFONO, SALARIO = pSALARIO ,ID_PUESTO = pIDPuesto, NO_FINCA = pIDFinca
    WHERE IDENTIFICACION_EMPLEADO = pID;    
    COMMIT;
  END EDITAR_EMPLEADO;
  
    PROCEDURE ELIMINAR_EMPLEADO (pID IN NUMBER) AS
  BEGIN
    DELETE FROM estudiante.EMPLEADOS 
  WHERE IDENTIFICACION_EMPLEADO = pID;
  COMMIT;
  END ELIMINAR_EMPLEADO;

END PROCE_EMPLEADO;
