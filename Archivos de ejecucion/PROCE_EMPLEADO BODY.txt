create or replace NONEDITIONABLE PACKAGE BODY PROCE_EMPLEADO AS

  PROCEDURE AGREGAR_EMPLEADO (pID IN NUMBER, pNOMBRE IN VARCHAR2,pAPELLIDO IN VARCHAR2, pEMAIL IN VARCHAR2,pTELEFONO IN NUMBER,pSALARIO IN NUMBER,pIDPuesto IN NUMBER,pIDFinca IN NUMBER) AS
  BEGIN
    INSERT INTO estudiante.EMPLEADOS (IDENTIFICACION_EMPLEADO,NOMBRE_EMPLEADO,APELLIDO_EMPLEADO,EMAIL,TELEFONO,SALARIO,ID_PUESTO,NO_FINCA)
    VALUES (pID,pnombre,papellido,pemail,ptelefono,psalario,pidpuesto,pidfinca);
    COMMIT;
  END AGREGAR_EMPLEADO;

  PROCEDURE BUSCAR_EMPLEADO (cur_EMPLEADOS OUT SYS_REFCURSOR) AS
  BEGIN
    OPEN cur_empleados FOR SELECT * FROM estudiante.EMPLEADOS;
  END BUSCAR_EMPLEADO;
  
   PROCEDURE EDITAR_EMPLEADO (pID IN NUMBER, pNOMBRE IN VARCHAR2,pAPELLIDO IN VARCHAR2, pEMAIL IN VARCHAR2,pTELEFONO IN NUMBER,pSALARIO IN NUMBER,pIDPuesto IN NUMBER,pIDFinca IN NUMBER) AS
  BEGIN
    UPDATE estudiante.EMPLEADOS 
    SET IDENTIFICACION_EMPLEADO = pID, NOMBRE_EMPLEADO = pNOMBRE, APELLIDO_EMPLEADO = pAPELLIDO, EMAIL = pEMAIL, TELEFONO = pTELEFONO, SALARIO = pSALARIO ,ID_PUESTO = pIDPuesto, NO_FINCA = pIDFinca
    WHERE IDENTIFICACION_EMPLEADO = pID;    
    COMMIT;
  END EDITAR_EMPLEADO;
  
    PROCEDURE ELIMINAR_EMPLEADO (pID IN NUMBER) AS
  BEGIN
    DELETE FROM estudiante.EMPLEADOS 
  WHERE IDENTIFICACION_EMPLEADO = pID;
  COMMIT;
  END ELIMINAR_EMPLEADO;

-- CURSORES
PROCEDURE LISTAR_EMP_EMAILS(cur_empleados OUT SYS_REFCURSOR) AS
BEGIN
    OPEN cur_empleados FOR
        SELECT NOMBRE_EMPLEADO, APELLIDO_EMPLEADO, EMAIL
        FROM EMPLEADOS
        ORDER BY NOMBRE_EMPLEADO;
END LISTAR_EMP_EMAILS;

PROCEDURE EMPLEADO_MAYOR_SALARIO(cur_empleados OUT SYS_REFCURSOR) AS
BEGIN
    OPEN cur_empleados FOR
        SELECT NOMBRE_EMPLEADO, SALARIO
        FROM EMPLEADOS
        WHERE SALARIO = (SELECT MAX(SALARIO) FROM EMPLEADOS);
END EMPLEADO_MAYOR_SALARIO;

PROCEDURE MOSTRAR_CONTACTO_EMPLEADOS(cur_contacto OUT SYS_REFCURSOR) AS
BEGIN
    OPEN cur_contacto FOR
        SELECT NOMBRE_COMPLETO, CONTACTO
        FROM VISTA_EMPLEADOS_CONTACTO;
END MOSTRAR_CONTACTO_EMPLEADOS;

PROCEDURE EMPLEADOS_X_PUESTO(p_cursor OUT SYS_REFCURSOR) AS
BEGIN
    OPEN p_cursor FOR
    SELECT * FROM VISTA_EMPLEADOS_PUESTO;
END EMPLEADOS_X_PUESTO;

  --MÁS PROCEDIMIENTOS
-- Obtener empleados con salario mayor a un valor dado
PROCEDURE EMPLEADOS_SALARIO_ALTO(pSALARIO_MIN IN NUMBER, cur_EMPLEADOS OUT SYS_REFCURSOR) AS
BEGIN
    OPEN cur_EMPLEADOS FOR
        SELECT * FROM EMPLEADOS
        WHERE SALARIO > pSALARIO_MIN;
END EMPLEADOS_SALARIO_ALTO;

-- Contar empleados por puesto (usando parámetro OUT)
PROCEDURE CONTAR_EMPLEADOS_POR_PUESTO(pID_PUESTO IN NUMBER, pCANTIDAD OUT NUMBER) AS
BEGIN
    SELECT COUNT(*) INTO pCANTIDAD
    FROM EMPLEADOS
    WHERE ID_PUESTO = pID_PUESTO;
END CONTAR_EMPLEADOS_POR_PUESTO;

-- Eliminar empleados con salario menor al límite
PROCEDURE ELIMINAR_EMPLEADOS_BAJO_SALARIO(pSALARIO_LIMITE IN NUMBER) AS
BEGIN
    DELETE FROM EMPLEADOS
    WHERE SALARIO < pSALARIO_LIMITE;
    COMMIT;
END ELIMINAR_EMPLEADOS_BAJO_SALARIO;

-- Buscar empleados cuyo nombre contenga un texto dado
PROCEDURE BUSCAR_EMPLEADOS_NOMBRE(pNOMBRE IN VARCHAR2, cur_EMPLEADOS OUT SYS_REFCURSOR) AS
BEGIN
    OPEN cur_EMPLEADOS FOR
        SELECT * FROM EMPLEADOS
        WHERE LOWER(NOMBRE_EMPLEADO) LIKE '%' || LOWER(pNOMBRE) || '%';
END BUSCAR_EMPLEADOS_NOMBRE;


END PROCE_EMPLEADO;
