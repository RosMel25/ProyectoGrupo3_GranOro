-- PUESTO

CREATE VIEW VISTA_PUESTO AS
SELECT ID_PUESTO, NOMBRE_PUESTO, MIN_SALARIO, MAX_SALARIO
FROM PUESTO;

--Diferencia entre salarios mínimos y máximos
CREATE VIEW vista_puestos_rangos AS
SELECT ID_PUESTO, NOMBRE_PUESTO, 
       MAX_SALARIO - MIN_SALARIO AS DIFERENCIA_SALARIAL
FROM PUESTO;

--Muestra la cantidad de empleados de cada puesto
CREATE VIEW VISTA_PUESTOS_EMPLEADOS AS
SELECT p.ID_PUESTO, p.NOMBRE_PUESTO, COUNT(e.IDENTIFICACION_EMPLEADO) AS CANTIDAD_EMPLEADOS
FROM PUESTO p
LEFT JOIN EMPLEADOS e ON p.ID_PUESTO = e.ID_PUESTO
GROUP BY p.ID_PUESTO, p.NOMBRE_PUESTO;

--Se muestra los salarios en moneda local
CREATE VIEW VISTA_PUESTOS_MONEDA AS
SELECT ID_PUESTO AS CODIGO, 
       NOMBRE_PUESTO AS PUESTO, 
       TO_CHAR(MIN_SALARIO, 'L9,999,999') AS SALARIO_MINIMO, 
       TO_CHAR(MAX_SALARIO, 'L9,999,999') AS SALARIO_MAXIMO
FROM PUESTO;

-- EMPLEADOS

CREATE VIEW VISTA_EMPLEADOS AS
SELECT IDENTIFICACION_EMPLEADO, NOMBRE_EMPLEADO, APELLIDO_EMPLEADO, EMAIL, TELEFONO, SALARIO
FROM EMPLEADOS;

--Muestra el nombre del puesto en lugar del ID del empleado
CREATE VIEW VISTA_EMPLEADOS_PUESTO AS
SELECT e.IDENTIFICACION_EMPLEADO, e.NOMBRE_EMPLEADO, e.APELLIDO_EMPLEADO, 
       p.NOMBRE_PUESTO, e.SALARIO
FROM EMPLEADOS e
JOIN PUESTO p ON e.ID_PUESTO = p.ID_PUESTO;

--Empleados que ganan más de 1.000.000
CREATE VIEW VISTA_EMPLEADOS_SALARIOALTO AS
SELECT IDENTIFICACION_EMPLEADO, NOMBRE_EMPLEADO, APELLIDO_EMPLEADO, SALARIO
FROM EMPLEADOS
WHERE SALARIO > 1000000;

--Muestra la finca de cada empleado
CREATE VIEW VISTA_EMPLEADOS_FINCA AS
SELECT e.IDENTIFICACION_EMPLEADO, e.NOMBRE_EMPLEADO, e.APELLIDO_EMPLEADO, 
       e.NO_FINCA
FROM EMPLEADOS e;

--Muestra la finca donde trabaja cada empleado
CREATE VIEW VISTA_EMPLEADOS_CON_FINCA AS
SELECT e.IDENTIFICACION_EMPLEADO, e.NOMBRE_EMPLEADO, e.APELLIDO_EMPLEADO, 
       f.NOMBRE_FINCA, e.SALARIO
FROM EMPLEADOS e
JOIN FINCA f ON e.NO_FINCA = f.NO_FINCA;

--Combinación nombre y apellido en una sola columna y también muestra el teléfono
CREATE VIEW VISTA_EMPLEADOS_CONTACTO AS
SELECT IDENTIFICACION_EMPLEADO, 
       NOMBRE_EMPLEADO || ' ' || APELLIDO_EMPLEADO AS NOMBRE_COMPLETO, 
       EMAIL, 
       'Teléfono: ' || TELEFONO AS CONTACTO
FROM EMPLEADOS;
