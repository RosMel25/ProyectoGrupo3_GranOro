CREATE OR REPLACE PACKAGE BODY PROCE_VEHICULOS AS 

    PROCEDURE AGREGAR_VEHICULO(
        pPlaca IN VARCHAR2, 
        pMarca IN VARCHAR2, 
        pModelo IN VARCHAR2, 
        pValorFiscal IN DECIMAL, 
        pNoFinca IN NUMBER, 
        pIdentificacionEmpleado IN NUMBER, 
        pIdProveedor IN NUMBER
    ) AS 
    BEGIN 
        INSERT INTO VEHICULO (Placa_Vehiculo, Marca_Vehiculo, Modelo_Vehiculo, Valor_Fiscal, No_Finca, Identificacion_Empleado, Id_Proveedor) 
        VALUES (pPlaca, pMarca, pModelo, pValorFiscal, pNoFinca, pIdentificacionEmpleado, pIdProveedor); 
        COMMIT; 
    END AGREGAR_VEHICULO;

    PROCEDURE EDITAR_VEHICULO(
        pPlaca IN VARCHAR2, 
        pMarca IN VARCHAR2, 
        pModelo IN VARCHAR2, 
        pValorFiscal IN DECIMAL, 
        pNoFinca IN NUMBER, 
        pIdentificacionEmpleado IN NUMBER, 
        pIdProveedor IN NUMBER
    ) AS 
    BEGIN 
        UPDATE VEHICULO 
        SET Marca_Vehiculo = pMarca, 
            Modelo_Vehiculo = pModelo, 
            Valor_Fiscal = pValorFiscal, 
            No_Finca = pNoFinca, 
            Identificacion_Empleado = pIdentificacionEmpleado, 
            Id_Proveedor = pIdProveedor 
        WHERE Placa_Vehiculo = pPlaca; 
        COMMIT; 
    END EDITAR_VEHICULO;

    PROCEDURE ELIMINAR_VEHICULO(pPlaca IN VARCHAR2) AS 
    BEGIN 
        DELETE FROM VEHICULO WHERE Placa_Vehiculo = pPlaca; 
        COMMIT; 
    END ELIMINAR_VEHICULO;

    PROCEDURE BUSCAR_VEHICULO(pPlaca IN VARCHAR2, CUR_VEHICULO OUT SYS_REFCURSOR) AS 
    BEGIN 
        OPEN CUR_VEHICULO FOR 
        SELECT * FROM VEHICULO WHERE Placa_Vehiculo = pPlaca; 
    END BUSCAR_VEHICULO;

    PROCEDURE LISTAR_VEHICULOS(CUR_VEHICULOS OUT SYS_REFCURSOR) AS 
    BEGIN 
        OPEN CUR_VEHICULOS FOR 
        SELECT * FROM VEHICULO; 
    END LISTAR_VEHICULOS;

END PROCE_VEHICULOS;
