CREATE OR REPLACE
PACKAGE BODY PROCE_FINCAS AS

  PROCEDURE AGREGAR_FINCA(pID IN NUMBER, pNOMBRE IN VARCHAR2, pTAMAÑO IN NUMBER, pUBICACION IN VARCHAR2) AS
  BEGIN
    INSERT INTO estudiante.FINCA (NO_FINCA,NOMBRE_FINCA,TAMAÑO_FINCA,UBICACION_FINCA)
    VALUES (pID,pnombre,ptamaño,pubicacion);
    COMMIT;
  END AGREGAR_FINCA;

  PROCEDURE EDITAR_FINCA (pID IN NUMBER, pNOMBRE IN VARCHAR2, pTAMAÑO IN NUMBER, pUBICACION IN VARCHAR2) AS
  BEGIN
    UPDATE estudiante.FINCA
    SET NO_FINCA = pID,
        NOMBRE_FINCA = pnombre,
        TAMAÑO_FINCA = ptamaño,
        UBICACION_FINCA = pubicacion
    WHERE NO_FINCA = pid;
    COMMIT;
  END EDITAR_FINCA;

  PROCEDURE ELIMINAR_FINCA (pID IN NUMBER) AS
  BEGIN
    DELETE FROM estudiante.FINCA
    WHERE NO_FINCA = pid;
    COMMIT;
  END ELIMINAR_FINCA;

  PROCEDURE BUSCAR_FINCA (pID IN NUMBER) AS

  BEGIN
        DELETE FROM estudiante.FINCA
    WHERE NO_FINCA = pid;
    COMMIT;
  END BUSCAR_FINCA;

  PROCEDURE LISTAR_FINCA (CUR_FINCAS OUT SYS_REFCURSOR) AS
  BEGIN
    OPEN cur_fincas FOR SELECT * FROM estudiante.FINCA;
  END LISTAR_FINCA;

END PROCE_FINCAS;
