CREATE TABLE AUDITORIA_MANTENIMIENTO (
    ID_AUDITORIA         NUMBER GENERATED BY DEFAULT AS IDENTITY,
    ID_MANTENIMIENTO     INT,
    PLACA_VEHICULO       VARCHAR2(20),
    ACCION               VARCHAR2(10),
    FECHA_ACCION         DATE,
    USUARIO_BD           VARCHAR2(30)
);
/
CREATE OR REPLACE TRIGGER TRG_AUDIT_MANTENIMIENTO
AFTER INSERT OR UPDATE OR DELETE ON Mant_Vehiculos
FOR EACH ROW
BEGIN
    IF INSERTING THEN
        INSERT INTO AUDITORIA_MANTENIMIENTO (ID_MANTENIMIENTO, PLACA_VEHICULO, ACCION, FECHA_ACCION, USUARIO_BD)
        VALUES (:NEW.Id_Mantenimiento, :NEW.Placa_Vehiculo, 'INSERT', SYSDATE, USER);

    ELSIF UPDATING THEN
        INSERT INTO AUDITORIA_MANTENIMIENTO (ID_MANTENIMIENTO, PLACA_VEHICULO, ACCION, FECHA_ACCION, USUARIO_BD)
        VALUES (:NEW.Id_Mantenimiento, :NEW.Placa_Vehiculo, 'UPDATE', SYSDATE, USER);

    ELSIF DELETING THEN
        INSERT INTO AUDITORIA_MANTENIMIENTO (ID_MANTENIMIENTO, PLACA_VEHICULO, ACCION, FECHA_ACCION, USUARIO_BD)
        VALUES (:OLD.Id_Mantenimiento, :OLD.Placa_Vehiculo, 'DELETE', SYSDATE, USER);
    END IF;
END;
/
