CREATE OR REPLACE
PACKAGE BODY PROCE_PUESTO AS

  PROCEDURE AGREGAR_PUESTO (pID IN NUMBER, pNOMBRE_PUESTO IN VARCHAR2,pMIN_SALARIO IN NUMBER,pMAX_SALARIO IN NUMBER) AS
  BEGIN    
    INSERT INTO estudiante.PUESTO (ID_PUESTO,NOMBRE_PUESTO,MIN_SALARIO,MAX_SALARIO)
    VALUES (pID,pnombre_puesto,pmin_salario,pmax_salario);
    COMMIT;
  END AGREGAR_PUESTO;

  PROCEDURE EDITAR_PUESTO (pID IN NUMBER, pNOMBRE_PUESTO IN VARCHAR2,pMIN_SALARIO IN NUMBER,pMAX_SALARIO IN NUMBER) AS
  BEGIN
    UPDATE estudiante.PUESTO
    SET ID_PUESTO = pID, NOMBRE_PUESTO = pnombre_puesto, MIN_SALARIO = pmin_salario, MAX_SALARIO = pmax_salario
    WHERE ID_PUESTO = pID;
    COMMIT;
  END EDITAR_PUESTO;

  PROCEDURE ELIMINAR_PUESTO (pID IN NUMBER) AS
  BEGIN
    DELETE FROM estudiante.PUESTO
  WHERE ID_PUESTO = pID;
  COMMIT;
  END ELIMINAR_PUESTO;

  PROCEDURE BUSCAR_PUESTO (cur_PUESTOS OUT SYS_REFCURSOR) AS
  BEGIN
    OPEN cur_puestos FOR SELECT * FROM estudiante.PUESTO;
  END BUSCAR_PUESTO;

END PROCE_PUESTO;
