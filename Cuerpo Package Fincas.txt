create or replace NONEDITIONABLE PACKAGE BODY PROCE_FINCAS AS

  PROCEDURE AGREGAR (pID IN NUMBER, pNOMBRE IN VARCHAR2, pTAMAÑO IN NUMBER, pUBICACION IN VARCHAR2) AS
  BEGIN
    INSERT INTO estudiante.FINCA (NO_FINCA,NOMBRE_FINCA,TAMAÑO_FINCA,UBICACION_FINCA)
    VALUES (pID,pnombre,ptamaño,pubicacion);
    COMMIT;
  END AGREGAR;

  PROCEDURE LISTAR (cur_FINCAS OUT SYS_REFCURSOR) AS
  BEGIN
    OPEN cur_fincas FOR SELECT * FROM estudiante.FINCA;
  END LISTAR;

END PROCE_FINCAS;
